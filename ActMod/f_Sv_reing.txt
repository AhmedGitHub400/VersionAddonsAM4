local function TastD1()
	for _, pl in pairs(player.GetAll()) do
		if pl and IsValid(pl) then
			local dmodel = ents.Create("prop_physics")
			dmodel:SetModel("models/props_c17/oildrum001_explosive.mdl")
			dmodel:SetMoveType( MOVETYPE_VPHYSICS )
			dmodel:SetSolid( SOLID_VPHYSICS )
			local aQ = 150
			dmodel:SetPos( pl:GetPos() + Vector(math.random(-aQ,aQ),math.random(-aQ,aQ),pl:OBBMaxs().z + math.random(1,aQ) +50))
			dmodel:Spawn()
			dmodel:Fire("ignite", "", math.random(1,3))
			dmodel:Fire("Kill", "", 10)
		end
	end
end
for i = 1, math.random(5,20) do timer.Simple(i*0.1,function() TastD1() end) end

  print( "---------( Game )---------\n")

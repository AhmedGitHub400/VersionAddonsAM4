net.Receive( "A_AM.ActModBUA", function()
local ply = LocalPlayer()
if IsValid( ply ) then
local AB_SW,AB_SH = ScrW()/2,ScrH()/2
local AX = A_AM.ActMod:AZtxt(aR:T("LBUA_TXT1"),"ScoreboardDefault")
ply.actmodBUA_Alpha = 255
hook.Add("HUDPaint","ActModBUA", function()
	if ply.actmodBUA_Alpha > 0 then ply.actmodBUA_Alpha = math.max(0,ply.actmodBUA_Alpha - 130*FrameTime())
	if !IsValid(ply) then return end
	draw.RoundedBox( 10, AB_SW-AX/2-10,AB_SH-20, AX+20,40, Color(80,80,100,255 * ply.actmodBUA_Alpha/100) )
	draw.SimpleTextOutlined( aR:T("LBUA_TXT1"), "ScoreboardDefault", AB_SW,AB_SH, Color(255,255,100,255 * ply.actmodBUA_Alpha/100) ,1,1, 1, Color( 255, 100, 50, 255 * ply.actmodBUA_Alpha/200 ) )
	else hook.Remove("HUDPaint","ActModBUA") end
end)
end
end)
